vector<vector<int>> generateMatrix(int n) {
        vector<vector<int>> result(n, vector<int>(n, 0));
        int Offset = 0;
        int row = 0, col = 0;
        int numCircle = n / 2;
        int value = 1;
        while(numCircle > 0)
        {
            row = Offset;
            col = Offset;

            for(col; col < n - Offset - 1; ++col)
                result[row][col] = value++;
            
            for(row; row < n - Offset - 1; ++row)
                result[row][col] = value++;

            for(col; col > Offset; --col)
                result[row][col] = value++;
            
            for(row; row > Offset; -- row)
                result[row][col] = value++;

            ++Offset;

            --numCircle;
        }
        if( n % 2 == 1)
        {
            result[Offset][Offset] = value;
        }
        return result;
    } ^xRKnNWtg
